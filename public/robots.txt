User-agent: *
Allow: /

# Private surfaces. These are all behind authentication and none of them is useful in an index:
# the admin console redirects a signed-out crawler to a login page, and the application flow is a
# six-step form belonging to one signed-in person. Crawling them only spends budget that should be
# going to the job postings.
Disallow: /admin
Disallow: /apply
Disallow: /portal

# EVERY LINE BELOW IS A FUNCTION INVOCATION THIS SITE WAS PAYING FOR AND GETTING NOTHING BACK.
#
# The site renders on Vercel with output: 'server', so there is no such thing as a free request:
# a crawler asking for a path it can never use still wakes the serverless function, still runs the
# middleware, and on many of these paths still opens a database connection. The Hobby plan meters
# that as Fluid Active CPU, and 3h31m of the 4h monthly allowance was gone when this was written.
#
# Nothing here is being hidden. robots.txt is not a security control and none of these paths relies
# on it for access — every one of them is gated server-side, which is exactly why crawling them is
# pointless: the crawler gets a redirect or a 403 and we get charged for producing it.

# The API is not a document. There is nothing here for a search index, and several of these
# endpoints run real queries before deciding they have nothing to say to an anonymous caller.
# The job feed lives here too, but the boards that consume it fetch it directly by URL and Google
# for Jobs reads the JobPosting JSON-LD embedded in /careers/<slug>, not this path.
Disallow: /api/

# Other people's consoles, on the same reasoning as /admin.
Disallow: /aquintutor/admin
Disallow: /hei/admin
Disallow: /founder
Disallow: /mail

# Onboarding and hiring flows belong to one named person mid-process.
Disallow: /onboarding
Disallow: /offer
Disallow: /interview
Disallow: /invite

# Per-record pages keyed by a token, an order or a slug. Unbounded to a crawler, meaningless
# without the record, and in the credential cases they are one identified person's certificate --
# which is a thing to verify on demand, not a thing to publish into a search index.
Disallow: /u/

Disallow: /f/
Disallow: /p/
Disallow: /r/
Disallow: /profile/
Disallow: /receipt/
Disallow: /surveys/
Disallow: /verify/
Disallow: /credentials/
